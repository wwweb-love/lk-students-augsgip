// Защита от NULL в числовых/булевых полях плана
Функция БезNULL(Значение, ЗначениеПоУмолчанию)
    
    Если Значение = NULL Тогда
        Возврат ЗначениеПоУмолчанию;
    КонецЕсли;
    
    Возврат Значение;

// Шапка учебного плана из выборки (общие алиасы полей обоих запросов)
Функция ЗаголовокПлана(Выборка)
    
    План = Новый Структура;
    План.Вставить("name",           Строка(Выборка.План));
    План.Вставить("code",           Строка(Выборка.КодПлана));
    План.Вставить("specialty",      Строка(Выборка.Специальность));
    План.Вставить("qualification",  Строка(Выборка.Квалификация));
    План.Вставить("form",           Строка(Выборка.ФормаОбучения));
    План.Вставить("start_year",     БезNULL(Выборка.ГодНачалаПодготовки, 0));
    План.Вставить("duration_years", БезNULL(Выборка.СрокОбученияЛет, 0));
    
    Возврат План;

// Дисциплины и распределение по семестрам для МАССИВА планов — двумя запросами на всё.
// ПланыДисциплин: Соответствие "план-ссылка -> Массив дисциплин" (наполняется).
Процедура ДобавитьДисциплиныПоПланам(МассивПланов, ПланыДисциплин)
    
    Если МассивПланов.Количество() = 0 Тогда
        Возврат;

    ЗапросДисциплины.Текст =

    |   С.Владелец КАК ПланСсылка,
    |   С.Ссылка КАК СтрокаСсылка,
    |   С.Дисциплина.Наименование КАК Дисциплина,
    |   С.ВсегоЗанятий КАК ВсегоЗанятий,
    |   С.МаксимальнаяНагрузка КАК МаксимальнаяНагрузка,
    |   С.ЛабораторныеРаботы КАК ЛабораторныеРаботы,

    |   С.СеминарныеЗанятия КАК СеминарныеЗанятия,
    |   С.СамостоятельнаяНагрузка КАК СамостоятельнаяНагрузка,
    |   С.ЭтоПрактика КАК ЭтоПрактика
    |ИЗ
    |   Справочник.СтрокиРУП КАК С
    |ГДЕ

    |УПОРЯДОЧИТЬ ПО
    |   С.Владелец,
    |   Дисциплина";

        КонецЕсли;

        Дисциплина.Вставить("practice_hours", БезNULL(Выборка.ПрактическиеРаботы, 0));
        Дисциплина.Вставить("seminar_hours",  БезNULL(Выборка.СеминарныеЗанятия, 0));
        Дисциплина.Вставить("self_hours",     БезNULL(Выборка.СамостоятельнаяНагрузка, 0));

        Дисциплина.Вставить("by_semester",    Новый Массив);
        
        МассивДисциплин.Добавить(Дисциплина);
        СоответствиеСтрок.Вставить(Выборка.СтрокаСсылка, Дисциплина);
        
    КонецЦикла;
    
    // === 2. ЧАСЫ ПО СЕМЕСТРАМ ДЛЯ ТЕХ ЖЕ ПЛАНОВ ===
    ЗапросСеместры = Новый Запрос;
    ЗапросСеместры.Текст =

    |   Р.Ссылка КАК СтрокаСсылка,
    |   ПРЕДСТАВЛЕНИЕ(Р.Семестр) КАК Семестр,
    |   Р.Всего КАК Всего,

        КонецЕсли;
        
        СтрокаСеместра = Новый Структура;

// Учебный план одной группы
Функция УчебныйПланГруппы(ИмяГруппы)
    
    Если ИмяГруппы = Неопределено ИЛИ НЕ ЗначениеЗаполнено(ИмяГруппы) Тогда
        Возврат ОтветJSON(400, Новый Структура("error", "Не указана группа (group_name)"));
    КонецЕсли;
    
    // === 1. ГРУППА -> УЧЕБНЫЙ ПЛАН ===
    ЗапросГруппа = Новый Запрос;
    ЗапросГруппа.Текст = 
    "ВЫБРАТЬ
    |   Г.Наименование КАК Группа,
    |   Г.Курс КАК Курс,
    |   Г.ГодПоступления КАК ГодПоступления,
    |   Г.РабочийУчебныйПлан КАК ПланСсылка,
    |   Г.РабочийУчебныйПлан.Наименование КАК План,
    |   Г.РабочийУчебныйПлан.Код КАК КодПлана,
    |   Г.РабочийУчебныйПлан.Специальность.Наименование КАК Специальность,
    |   Г.РабочийУчебныйПлан.Квалификация.Наименование КАК Квалификация,
    |   ПРЕДСТАВЛЕНИЕ(Г.РабочийУчебныйПлан.ФормаОбучения) КАК ФормаОбучения,
    |   Г.РабочийУчебныйПлан.ГодНачалаПодготовки КАК ГодНачалаПодготовки,
    |   Г.РабочийУчебныйПлан.НормативныйСрокОбученияЛет КАК СрокОбученияЛет
    |ИЗ
    |   Справочник.УчебныеГруппы КАК Г
    |ГДЕ
    |   Г.Наименование = &Группа";
    ЗапросГруппа.УстановитьПараметр("Группа", ИмяГруппы);
    
    ВыборкаГруппа = ЗапросГруппа.Выполнить().Выбрать();
    
    Если НЕ ВыборкаГруппа.Следующий() Тогда
        Возврат ОтветJSON(404, Новый Структура("error", "Группа не найдена: " + ИмяГруппы));
    КонецЕсли;
    
    Если ВыборкаГруппа.ПланСсылка = Справочники.РабочиеУчебныеПланы.ПустаяСсылка() Тогда
        Возврат ОтветJSON(404, Новый Структура("error", "Для группы не назначен учебный план"));
    КонецЕсли;
    
    // === 2. СБОРКА ОТВЕТА ===
    Ответ = Новый Структура;
    Ответ.Вставить("group_name",     Строка(ВыборкаГруппа.Группа));
    Ответ.Вставить("course",         БезNULL(ВыборкаГруппа.Курс, 0));
    Ответ.Вставить("admission_year", БезNULL(ВыборкаГруппа.ГодПоступления, 0));
    Ответ.Вставить("plan",           ЗаголовокПлана(ВыборкаГруппа));
    
    МассивПланов = Новый Массив;
    МассивПланов.Добавить(ВыборкаГруппа.ПланСсылка);
    
    МассивДисциплин = Новый Массив;
    ПланыДисциплин = Новый Соответствие;
    ПланыДисциплин.Вставить(ВыборкаГруппа.ПланСсылка, МассивДисциплин);
    
    ДобавитьДисциплиныПоПланам(МассивПланов, ПланыДисциплин);
    
    Ответ.Вставить("disciplines", МассивДисциплин);
    
    Возврат ОтветJSON(200, Ответ);
    
КонецФункции

    |   НЕ Г.ПометкаУдаления
    |   И НЕ Г.ЭтоГруппа
    |   И Г.РабочийУчебныйПлан <> ЗНАЧЕНИЕ(Справочник.РабочиеУчебныеПланы.ПустаяСсылка)
    |УПОРЯДОЧИТЬ ПО
    |   Г.Наименование";
    
    МассивГрупп = Новый Массив;
    МассивПланов = Новый Массив;
    ПланыДисциплин = Новый Соответствие;
    
    Выборка = ЗапросГруппы.Выполнить().Выбрать();

    // === 2. ДИСЦИПЛИНЫ И СЕМЕСТРЫ ВСЕХ ПЛАНОВ (2 запроса на всё) ===
