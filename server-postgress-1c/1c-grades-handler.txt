//     "group_name":   "9С-254",
//     "grades": [

//         "semester":   2,                     // число или null
//         "date":       "2026-01-15",          // YYYY-MM-DD или null
//         "work_type":  "Экзамен",            // Экзамен/Зачёт/ЛР/ПР/КР…
//         "grade":      5,                     // 2..5; null, если зачёт
//         "passed":     null                   // true/false только для зачётов; null для 5-балльных
//       }
//     ]
//   }

// ⚠️ ВАЖНО: в дампах конфигурации (1c-objects.txt) объектов журнала оценок нет,
// поэтому ниже обработчик написан ОБЩИМ СПОСОБОМ: он ищет оценки в наиболее
// типичных местах учебной части и объединяет их. Поля запросов помечены
// комментарием «АДАПТИРОВАТЬ» — сверьте их с реальной конфигурацией базы 1С
// (типичные варианты: РегистрСведений.Успеваемость, РегистрСведений.Оценки,
// Документ.ЭкзаменационнаяВедомость.Оценки, Справочник.СведенияОбУспеваемости).

Функция GradesGET(Запрос)
    
    Возврат УспеваемостьВсехСтудентов();
    
КонецФункции

// Защита от NULL в числовых/булевых полях (как в 1c-plan-handlers.txt)
Функция БезNULL(Значение, ЗначениеПоУмолчанию)
    
    Если Значение = NULL Тогда
        Возврат ЗначениеПоУмолчанию;
    КонецЕсли;
    
    Возврат Значение;
    
КонецФункции

// СНИЛС в формате 11 цифр: «XXX-XXX-XXX XX» -> «XXXXXXXXXXX»
Функция СНИЛС11Цифр(Знач СтрСнилс)
    
    Дigits = "";
    Для Позиция = 1 По СтрДлина(СтрСнилс) Цикл
        Символ = Сред(СтрСнилс, Позиция, 1);
        Если Символ >= "0" И Символ <= "9" Тогда
            Дigits = Дigits + Символ;
        КонецЕсли;
    КонецЦикла;
    
    Если СтрДлина(Digits) <> 11 Тогда
        Возврат ""; // не сопоставим со студентом — пропуск на стороне приёмника
    КонецЕсли;
    
    Возврат Дigits;
    
КонецФункции

// Дата в формате YYYY-MM-DD (для JSON без посредников)
Функция ДатаJSON(ДатаЗначение)
    
    Если НЕ ЗначениеЗаполнено(ДатаЗначение) Тогда
        Возврат Неопределено;
    КонецЕсли;
    
    Возврат Формат(ДатаЗначение, "ДФ=yyyy-MM-dd");
    
КонецФункции

// ============================================================================
// СБОР ЖУРНАЛА ОЦЕНОК
// ============================================================================
// Логика: один проход по всем несданным/сданным работам студентов.
// Структура: Соответствие "СтудентСсылка -> Структура{ФИО, СНИЛС, Группа, МассивОценок}"
//
// АДАПТИРОВАТЬ №1 — ВЫБОРКА ОЦЕНОК.
// Замените текст запроса на реальный объект хранения оценок вашей конфигурации.
// Обязательные поля выборки: Студент, Дисциплина, Семестр, Дата, ВидРаботы, Оценка, Зачёт.
// ============================================================================
Функция УспеваемостьВсехСтудентов()
    
    // --- КАРТА СТУДЕНТОВ: СНИЛС + ФИО + группа (за один проход по учебным группам) ---
    ЗапросСтуденты = Новый Запрос;
    ЗапросСтуденты.Текст = 
    "ВЫБРАТЬ
    |   Ст.Ссылка КАК СтудентСсылка,
    |   Ст.Наименование КАК ФИО,
    |   Ст.СНИЛС КАК СНИЛС,
    |   Г.Наименование КАК Группа
    |ИЗ
    |   Справочник.Студенты КАК Ст           // АДАПТИРОВАТЬ: справочник студентов
    |   ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.УчебныеГруппы КАК Г
    |   ПО Ст.Группа = Г.Ссылка
    |ГДЕ
    |   НЕ Ст.ПометкаУдаления";
    
    КартаСтудентов = Новый Соответствие;
    
    ВыборкаСтуденты = ЗапросСтуденты.Выполнить().Выбрать();
    
    Пока ВыборкаСтуденты.Следующий() Цикл
        
        СНИЛС = СНИЛС11Цифр(Строка(ВыборкаСтуденты.СНИЛС));
        Если СНИЛС = "" Тогда
            Продолжить; // без СНИЛС студент не сопоставим с личным кабинетом
        КонецЕсли;
        
        Сведения = Новый Структура;
        Сведения.Вставить("student_name", Строка(ВыборкаСтуденты.ФИО));
        Сведения.Вставить("group_name",   Строка(ВыборкаСтуденты.Группа));
        Сведения.Вставить("grades",       Новый Массив);
        
        КартаСтудентов.Вставить(ВыборкаСтуденты.СтудентСсылка, Сведения);
        
    КонецЦикла;
    
    Если КартаСтудентов.Количество() = 0 Тогда
        Возврат ОтветJSON(200, Новый Массив);
    КонецЕсли;
    
    // --- ОЦЕНКИ ВСЕХ СТУДЕНТОВ ЗА ОДИН ЗАПРОС ---
    // АДАПТИРОВАТЬ №1: замените объект-источник и имена полей на реальные.
    // Типичный источник в конфигурациях колледжей — накопительный регистр сведений
    // с измерениями Студент/Дисциплина/Семестр и ресурсами Оценка/Зачёт.
    ЗапросОценки = Новый Запрос;
    ЗапросОценки.Текст = 
    "ВЫБРАТЬ
    |   О.Студент КАК Студент,
    |   ПРЕДСТАВЛЕНИЕ(О.Дисциплина) КАК Дисциплина,
    |   О.Семестр.Код КАК Семестр,
    |   О.Дата КАК Дата,
    |   ПРЕДСТАВЛЕНИЕ(О.ВидРаботы) КАК ВидРаботы,
    |   О.Оценка КАК Оценка,
    |   О.Зачёт КАК Зачёт
    |ИЗ
    |   РегистрСведений.Успеваемость КАК О     // АДАПТИРОВАТЬ: реальный регистр/табличная часть
    |ГДЕ
    |   О.Студент В (&Студенты)
    |УПОРЯДОЧИТЬ ПО
    |   О.Дата";
    
    МассивСтудентов = Новый Массив;
    Для Каждого КлючЗначение Из КартаСтудентов Цикл
        МассивСтудентов.Добавить(КлючЗначение.Ключ);
    КонецЦикла;
    
    ЗапросОценки.УстановитьПараметр("Студенты", МассивСтудентов);
    
    ВыборкаОценки = ЗапросОценки.Выполнить().Выбрать();
    
    Пока ВыборкаОценки.Следующий() Цикл
        
        Сведения = КартаСтудентов.Получить(ВыборкаОценки.Студент);
        Если Сведения = Неопределено Тогда
            Продолжить;
        КонецЕсли;
        
        Оценка25     = БезNULL(ВыборкаОценки.Оценка, 0);
        ФлагЗачёт    = ВыборкаОценки.Зачёт; // NULL / ИСТИНА / ЛОЖЬ (для 5-балльных — NULL)
        
        // Согласованный контракт: либо 2..5, либо зачёт/незачёт; прочее — мусор
        Если Оценка25 >= 2 И Оценка25 <= 5 Тогда
            JSONОценка    = Оценка25;
            JSONЗачёт     = Неопределено; // 5-балльная — флаг не нужен
        ИначеЕсли ФлагЗачёт = ИСТИНА ИЛИ ФлагЗачёт = ЛОЖЬ Тогда
            JSONОценка    = Неопределено; // зачёт — числа нет
            JSONЗачёт     = ФлагЗачёт;
        Иначе
            Продолжить; // пустая строка выгрузки
        КонецЕсли;
        
        СтрокаОценки = Новый Структура;
        СтрокаОценки.Вставить("discipline", Строка(ВыборкаОценки.Дисциплина));
        СтрокаОценки.Вставить("semester",   БезNULL(ВыборкаОценки.Семестр, Неопределено));
        СтрокаОценки.Вставить("date",       ДатаJSON(ВыборкаОценки.Дата));
        СтрокаОценки.Вставить("work_type",  Строка(ВыборкаОценки.ВидРаботы));
        СтрокаОценки.Вставить("grade",      JSONОценка);
        СтрокаОценки.Вставить("passed",     JSONЗачёт);
        
        Сведения.grades.Добавить(СтрокаОценки);
        
    КонецЦикла;
    
    // --- СБОРКА МАССИВА СТУДЕНТОВ (только у кого есть хоть одна оценка) ---
    МассивОтвета = Новый Массив;
    
    Для Каждого КлючЗначение Из КартаСтудентов Цикл
        
        СтудентСсылка = КлючЗначение.Ключ;
        Сведения      = КлючЗначение.Значение;
        
        Если Сведения.grades.Количество() = 0 Тогда
            Продолжить;
        КонецЕсли;
        
        СНИЛС = СНИЛС11Цифр(Строка(СтудентСсылка.СНИЛС)); // АДАПТИРОВАТЬ №2: реквизит СНИЛС
        
        СтудентJSON = Новый Структура;
        СтудентJSON.Вставить("snils",        СНИЛС);
        СтудентJSON.Вставить("student_name", Сведения.student_name);
        СтудентJSON.Вставить("group_name",   Сведения.group_name);
        СтудентJSON.Вставить("grades",       Сведения.grades);
        
        МассивОтвета.Добавить(СтудентJSON);
        
    КонецЦикла;
    
    Возврат ОтветJSON(200, МассивОтвета);
    
КонецФункции
